cmake_minimum_required(VERSION 3.20)
project(NeuralNetCppWineQuality LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Torch REQUIRED)
find_package(Python COMPONENTS Interpreter Development REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(wine_engine src/bindings.cpp src/wine_model.cpp)
target_include_directories(wine_engine PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(wine_engine PRIVATE "${TORCH_LIBRARIES}")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
if(MSVC)
  target_compile_options(wine_engine PRIVATE /W4 /wd4267 /wd4702)
endif()
